**Goal:** Add Brazilian Portuguese AMR as a third language branch to the ViSemPar pipeline, training on Little Prince PT-BR (~1,526 sentences) and evaluating on LPP-PT test + zero-shot AMRNews transfer.

**Architecture:** Mirror the existing `english_lpp` pattern exactly: download → prepare → format → configs → run pipeline. AMRNews requires preprocessing to strip `~e.N` alignment markers and handle `# ::tok-pt` headers. The six MTUP methods × 3 seeds run on LPP-PT test; AMRNews is zero-shot inference only (best seed checkpoint, no training).

**Tech Stack:** Python 3, existing `src/data_pipeline.py`, `src/inference.py`, `src/evaluate.py`, `src/aggregate_scores.py`; PyYAML configs; `git` for raw data vendoring.

## Task 2: Write `src/prepare_portuguese.py`

**Files:**
- Create: `src/prepare_portuguese.py`

This script: (1) reads AMR-BR LPP single file and 80/10/10 splits it; (2) reads AMRNews and uses its existing train/dev/test splits — keeping only the test split for zero-shot eval; (3) strips `~e.N` alignment markers from all AMR graphs; (4) normalises `# ::tok-pt` headers to `# ::snt`; (5) writes standard output files.

- [ ] **Step 1: Write the script**

```python
#!/usr/bin/env python3
"""Prepare Brazilian Portuguese AMR corpora for ViSemPar experiments.

Sources:
  AMR-BR (LPP):  rafaelanchieta/amr-br  amr_br-v1.0.txt   (1526 sentences)
  AMRNews:       nilc-nlp/AMR-BP        AMRNews/unsplit/amr.txt  (870 sentences)

Outputs (LPP):
  data/portuguese/little_prince/raw/train_amr_1.txt
  data/portuguese/little_prince/raw/train_amr_2.txt   (empty placeholder)
  data/portuguese/little_prince/raw/dev_ground_truth.txt
  data/portuguese/little_prince/raw/dev.txt
  data/portuguese/little_prince/raw/test_ground_truth.txt
  data/portuguese/little_prince/raw/test.txt

Outputs (news, zero-shot eval only):
  data/portuguese/news/raw/test_ground_truth.txt
  data/portuguese/news/raw/test.txt
"""

import re
import argparse
from pathlib import Path


ALIGN_RE = re.compile(r'~e\.\d+(?:,\d+)*')


def strip_alignment_markers(amr: str) -> str:
    """Remove ~e.N alignment suffixes from AMR concept/relation tokens."""
    return ALIGN_RE.sub('', amr)


def read_blocks(path: Path) -> list[str]:
    """Read AMR file into list of raw blocks (one block = one sentence)."""
    content = path.read_text(encoding='utf-8')
    blocks = []
    for block in content.strip().split('\n\n'):
        block = block.strip()
        if not block:
            continue
        # Must have at least one AMR line (starts with '(')
        has_amr = any(l.startswith('(') for l in block.split('\n'))
        if not has_amr:
            continue
        blocks.append(block)
    return blocks


def normalise_block(block: str) -> str:
    """Normalise a block: tok-pt → snt header, strip alignment markers."""
    lines = block.split('\n')
    out = []
    for line in lines:
        # Normalise ::tok-pt to ::snt
        line = re.sub(r'^# ::tok-pt\s+', '# ::snt ', line)
        # Drop other alignment/tok lines that aren't ::snt or ::id
        if line.startswith('#'):
            if re.match(r'^# ::(snt|id)\b', line):
                out.append(line)
            # Drop ::tok-en, ::alignments-*, ::save-date, etc.
        else:
            # AMR line — strip alignment markers
            out.append(strip_alignment_markers(line))
    return '\n'.join(out)


def write_blocks(path: Path, blocks: list[str]) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text('\n\n'.join(blocks) + '\n', encoding='utf-8')


def write_sentences(path: Path, blocks: list[str]) -> None:
    sentences = []
    for block in blocks:
        m = re.search(r'^# ::snt\s+(.+)$', block, flags=re.MULTILINE)
        if m:
            sentences.append(m.group(1).strip())
    path.write_text('\n'.join(sentences) + '\n', encoding='utf-8')


def split_lpp(blocks: list[str], train_ratio: float = 0.8, dev_ratio: float = 0.1):
    """Deterministic 80/10/10 split (no shuffle — preserves narrative order)."""
    n = len(blocks)
    train_end = int(n * train_ratio)
    dev_end = train_end + int(n * dev_ratio)
    return blocks[:train_end], blocks[train_end:dev_end], blocks[dev_end:]


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--lpp_raw', default='data/portuguese/little_prince/raw/amr_br_raw.txt')
    parser.add_argument('--news_raw', default='data/portuguese/news/raw/amrnews_raw.txt')
    parser.add_argument('--lpp_out', default='data/portuguese/little_prince/raw')
    parser.add_argument('--news_out', default='data/portuguese/news/raw')
    args = parser.parse_args()

    lpp_out = Path(args.lpp_out)
    news_out = Path(args.news_out)

    # ── LPP ──────────────────────────────────────────────────────────────────
    lpp_raw = read_blocks(Path(args.lpp_raw))
    lpp_blocks = [normalise_block(b) for b in lpp_raw]
    train_blocks, dev_blocks, test_blocks = split_lpp(lpp_blocks)

    write_blocks(lpp_out / 'train_amr_1.txt', train_blocks)
    write_blocks(lpp_out / 'train_amr_2.txt', [])          # empty placeholder
    write_blocks(lpp_out / 'dev_ground_truth.txt', dev_blocks)
    write_sentences(lpp_out / 'dev.txt', dev_blocks)
    write_blocks(lpp_out / 'test_ground_truth.txt', test_blocks)
    write_sentences(lpp_out / 'test.txt', test_blocks)

    print(f"LPP-PT  → train:{len(train_blocks)}  dev:{len(dev_blocks)}  test:{len(test_blocks)}")

    # ── AMRNews (test split only for zero-shot eval) ─────────────────────────
    news_raw = read_blocks(Path(args.news_raw))
    news_blocks = [normalise_block(b) for b in news_raw]

    # Use last 20% as held-out test (≈174 sentences); rest ignored for training
    n = len(news_blocks)
    news_test = news_blocks[int(n * 0.8):]

    write_blocks(news_out / 'test_ground_truth.txt', news_test)
    write_sentences(news_out / 'test.txt', news_test)

    print(f"AMRNews → test (zero-shot): {len(news_test)}")


if __name__ == '__main__':
    main()
```
